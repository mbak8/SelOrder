# Tworzymy projekt o nazwie 'client' w technologii React + TypeScript
npm create vite@latest client -- --template react-ts

# Wchodzimy do folderu
cd client

# Instalujemy biblioteki
npm install

# Instalujemy Axios (lepszy klient HTTP niż domyślny fetch)
npm install axios



-----------
Set-ExecutionPolicy -ExecutionPolicy RemoteSigned -Scope CurrentUser

npm run dev
